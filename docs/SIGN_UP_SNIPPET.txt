(defun most-common-word (text)
  (let ((counts (make-hash-table :test #'equal))
        (best ""))
    (loop for start = 0 then (1+ end)
          for end = (position #\Space text :start start)
          for word = (string-downcase (subseq text start end))
          do (when (> (incf (gethash word counts 0))
                      (gethash best counts 0))
               (setf best word))
          while end)
    (format nil "\"~a\" appears ~a times" best (gethash best counts))))
